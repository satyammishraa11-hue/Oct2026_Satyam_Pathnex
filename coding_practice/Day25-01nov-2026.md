# Day 25 — Continuous Monitoring and Alerts

## Ansible — Setup Prometheus Node Exporter for Monitoring

```yaml
  - name: Setup prometheus node exporter
    hosts: all
    become: yes
    tasks:
      - name: Install prometheus node exporter
        yum:
          name: start prometheus node exporter
          state: present
      - name: start prometheus node exporter service
        service:
          name: prometheus node-exporter
          state: started
          enabled: yes
```
## Terraform -  create cloudwatch alarm

resource "aws_cloudwatch_metric_alarm" "pathnex_cpu_alarm" {
    alarm_name        =  "pathnex-cpu-utilization-high"
    comparison_operator = "GreaterThanThresold"
    evaluation_periods  = "1"
    metric_name         = "CPUUtilization"
    namespace           = "AWS/EC2"
    period              = "60"
    statistic           = "Average"
    thresold            = "80"

    dimentions  = {
        Installed = aws_instance.pathnex_ec2.id
    }
    alarm_action = [aws_sns_topic.pathnex_alerts.arn]
}

resource "aws_sns_topic" "pathnex_alerts " {
    name  =  "pathnex_alerts"
}
